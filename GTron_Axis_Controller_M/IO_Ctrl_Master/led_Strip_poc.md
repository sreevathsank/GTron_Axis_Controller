bool SERCOM3_SPI_TransferSetup(SPI_TRANSFER_SETUP *setup, uint32_t spiSourceClock)
{
    uint32_t baudValue = 0U;

    bool statusValue = false;

    if(spiSourceClock == 0U)
    {
        /* Fetch Master Clock Frequency directly */
        spiSourceClock = SERCOM3_Frequency;
    }

    /* Disable the SPI Module */
    SERCOM3_REGS->SPIM.SERCOM_CTRLA &= ~(SERCOM_SPIM_CTRLA_ENABLE_Msk);


    /* Wait for synchronization */
    while((SERCOM3_REGS->SPIM.SERCOM_SYNCBUSY) != 0U)
    {
        /* Do nothing */
    }

    if(setup != NULL)
    {
        if (setup->clockFrequency <= spiSourceClock/2U)
        {
            baudValue = (spiSourceClock/(2U*(setup->clockFrequency))) - 1U;

            /* Set the lowest possible baud */
            if (baudValue >= 255U)
            {
                baudValue = 255U;
            }

            /* Selection of the Clock Polarity and Clock Phase */
            SERCOM3_REGS->SPIM.SERCOM_CTRLA &= ~(SERCOM_SPIM_CTRLA_CPOL_Msk | SERCOM_SPIM_CTRLA_CPHA_Msk);
            SERCOM3_REGS->SPIM.SERCOM_CTRLA |= (uint32_t)setup->clockPolarity | (uint32_t)setup->clockPhase;

            /* Selection of the Baud Value */
            SERCOM3_REGS->SPIM.SERCOM_BAUD = (uint8_t)baudValue;

            /* Selection of the Character Size */
            SERCOM3_REGS->SPIM.SERCOM_CTRLB &= ~SERCOM_SPIM_CTRLB_CHSIZE_Msk;
            SERCOM3_REGS->SPIM.SERCOM_CTRLB |= (uint32_t)setup->dataBits;

            while((SERCOM3_REGS->SPIM.SERCOM_SYNCBUSY) != 0U)

    /* Wait for synchronization */
    while((SERCOM3_REGS->SPIM.SERCOM_SYNCBUSY) != 0U)
    {
        /* Do nothing */
    }

{

    size_t rxCount = 0U;
    size_t dummySize = 0U;
    size_t receivedData;
    uint32_t dataBits;
    bool isSuccess = false;

    /* Verify the request */
    if(((txSize > 0U) && (pTransmitData != NULL)) || ((rxSize > 0U) && (pReceiveData != NULL)))
    {
        dataBits = SERCOM3_REGS->SPIM.SERCOM_CTRLB & SERCOM_SPIM_CTRLB_CHSIZE_Msk;

        if(dataBits != (uint32_t)SPI_DATA_BITS_8)
        {
            /* For 9-bit transmission, the txSize and rxSize must be an even number. */
            if(((txSize > 0U) && ((txSize & 0x01U) != 0U)) || ((rxSize > 0U) && ((rxSize & 0x01U) != 0U)))
            {
                return isSuccess;
            }
        }

        if(pTransmitData == NULL)
        {
            txSize = 0U;
        }

        if(pReceiveData == NULL)
        {
            rxSize = 0U;
        }

        /* Flush out any unread data in SPI DATA Register from the previous transfer */
        while((SERCOM3_REGS->SPIM.SERCOM_INTFLAG & SERCOM_SPIM_INTFLAG_RXC_Msk) == SERCOM_SPIM_INTFLAG_RXC_Msk)
        {
            receivedData = SERCOM3_REGS->SPIM.SERCOM_DATA;
        }

        SERCOM3_REGS->SPIM.SERCOM_STATUS |= (uint16_t)SERCOM_SPIM_STATUS_BUFOVF_Msk;

        SERCOM3_REGS->SPIM.SERCOM_INTFLAG |= (uint8_t)SERCOM_SPIM_INTFLAG_ERROR_Msk;

        if(rxSize > txSize)
        {
            dummySize = rxSize - txSize;
        }

        if(dataBits != (uint32_t)SPI_DATA_BITS_8)
        {
            rxSize >>= 1U;
            txSize >>= 1U;
            dummySize >>= 1U;
        }

        /* Make sure DRE is empty */
        while((SERCOM3_REGS->SPIM.SERCOM_INTFLAG & SERCOM_SPIM_INTFLAG_DRE_Msk) != SERCOM_SPIM_INTFLAG_DRE_Msk)
        {
            /* Do nothing */
        }

        while((txCount != txSize) || (dummySize != 0U))
        {
            if(txCount != txSize)
            {
                if(dataBits == (uint32_t)SPI_DATA_BITS_8)
                {
                    SERCOM3_REGS->SPIM.SERCOM_DATA = ((uint8_t*)pTransmitData)[txCount];
                    txCount++;
                }
                else
                {
                    SERCOM3_REGS->SPIM.SERCOM_DATA = ((uint16_t*)pTransmitData)[txCount] & SERCOM_SPIM_DATA_Msk;
                    txCount++;
                }
            }
            else if(dummySize > 0U)
            {
                if(dataBits == (uint32_t)SPI_DATA_BITS_8)
                {
                    SERCOM3_REGS->SPIM.SERCOM_DATA = 0xFFU;
                }
                else
                {
                    SERCOM3_REGS->SPIM.SERCOM_DATA = 0xFFFFU & SERCOM_SPIM_DATA_Msk;
                }

                dummySize--;
            }
            else
            {
                /* Do nothing */
            }

            if(rxSize == 0U)
            {
                /* For transmit only request, wait for DRE to become empty */
                while((SERCOM3_REGS->SPIM.SERCOM_INTFLAG & SERCOM_SPIM_INTFLAG_DRE_Msk) != SERCOM_SPIM_INTFLAG_DRE_Msk)
                {
                    /* Do nothing */
                }
            }
            else
            {
                /* If data is read, wait for the Receiver Data Register to become full */
                while((SERCOM3_REGS->SPIM.SERCOM_INTFLAG & SERCOM_SPIM_INTFLAG_RXC_Msk) != SERCOM_SPIM_INTFLAG_RXC_Msk)
                {
                    /* Do nothing */
                }

                if(rxCount < rxSize)
                {
                    if(dataBits == (uint32_t)SPI_DATA_BITS_8)
                    {
                        ((uint8_t*)pReceiveData)[rxCount++] = (uint8_t)receivedData;
                    }
                    else
                    {
                        ((uint16_t*)pReceiveData)[rxCount++] = (uint16_t)(receivedData & SERCOM_SPIM_DATA_Msk);
                    }
                }
            }
        }

        /* Make sure no data is pending in the shift register */
        while((SERCOM3_REGS->SPIM.SERCOM_INTFLAG & SERCOM_SPIM_INTFLAG_TXC_Msk) != SERCOM_SPIM_INTFLAG_TXC_Msk)
        {
            /* Do nothing */
        }

        isSuccess = true;
    }

    return isSuccess;
}

## MODEL ✨

Using the SPI MOSI driver with **FreeRTOS** is significantly safer than bit-banging, but there is still one critical risk you must handle: **Context Switching**.

Here is the breakdown of how FreeRTOS affects this specific driver and how to handle it.

### 1. The Good News: Bit Timing is Safe
Unlike the bit-banging version, **you do NOT need to disable interrupts** to maintain the integrity of the 0s and 1s.
*   **Bit-Banging:** A single interrupt firing in the middle of a bit destroys the timing (20ns precision required).
*   **SPI:** The hardware peripheral generates the pulses. Even if an interrupt fires, the current byte will finish transmitting perfectly.

### 2. The Risk: Frame Splitting (Buffer Underrun)
The risk in an RTOS is **Preemption**.
*   The SPI peripheral has a buffer (Data Register). It needs to be fed a new byte every **~2.67µs** (at 3MHz).
*   Your current `LED_Show_SPI` function uses a **Blocking Polling Loop** (`while(!DRE)...`).
*   **Scenario:**
    1.  The LED task sends 50 bytes.
    2.  FreeRTOS triggers a Context Switch to a higher-priority task (e.g., USB or WiFi).
    3.  The LED task is paused for **100µs**.
    4.  The SPI peripheral finishes the current byte and runs out of data. The MOSI line goes Idle (Low).
    5.  The LED Strip sees the line Low for > 50µs and thinks: **"RESET! End of Frame."**
    6.  The LED task resumes and sends the rest of the bytes.
    7.  **Result:** The strip displays the first half of the LEDs, flashes, and treats the second half as a new frame starting from LED 0.

### 3. The CPU Usage
The current driver is **Blocking**. If you have 100 LEDs:
*   Transmission time: ~3.5 ms.
*   During this 3.5 ms, the CPU is spinning in a `while` loop inside `LED_Show_SPI`.
*   This is effectively "100% CPU load" for that duration, preventing lower-priority tasks from running.

---

### Solution A: The Quick Fix (Critical Section)
To prevent FreeRTOS from switching tasks during the transmission, wrap the send command in a **Critical Section** or **Scheduler Suspension**.

In your FreeRTOS task:
```c
#include "FreeRTOS.h"
#include "task.h"

void LED_Task(void *pvParameters) {
    LED_Driver_SPI_Init();
    
    while(1) {
        // ... update colors ...
        
        // ENTER CRITICAL: Disable Context Switches
        taskENTER_CRITICAL(); 
        
        LED_Show_SPI(); // The loop runs uninterrupted
        
        taskEXIT_CRITICAL();
        // EXIT CRITICAL
        
        vTaskDelay(pdMS_TO_TICKS(30));
    }
}
```
*   **Pros:** Guarantees no flickering/split frames. Simple to implement.
*   **Cons:** Freezes the entire system (including high-priority tasks) for the duration of the LED update (e.g., 3ms).

### Solution B: The "Pro" Way (DMA)
If your system cannot afford to freeze for 3ms (e.g., you have strict audio or motor control loops), you must use **DMA (Direct Memory Access)**.
1.  Configure a DMA Channel to transfer the `spi_buffer` to the `SERCOM3 DATA` register.
2.  Trigger the DMA transfer.
3.  The CPU can now go to sleep or run other tasks.
4.  DMA keeps the SPI fed instantly, immune to context switches.

**For a POC or typical UI LED usage, Solution A (Critical Section) is usually standard practice.**

*   **Trigger Source (`SERCOM3_Transmit`):** This ensures the DMA only moves data when the SPI hardware is actually ready for the next byte.
*   **Trigger Action (`One Beat...`):** Since SPI is a byte-by-byte protocol, moving one "beat" (8-bits) per request prevents buffer overflows.
*   **Destination Address Mode (`Fixed`):** This is critical. It ensures the DMA always "points" at the SPI Data Register and doesn't try to write into random memory locations after the first byte.
*   **Source Address Mode (`Increment`):** This allows the DMA to "walk" through your `spi_buffer` array bit-by-bit.
*   **Beat Size (`8-bit`):** Matches the SPI character size.

**One tip for your Production Code:** 
Since you have **Enable Interrupt** checked, the DMAC will generate an interrupt when the entire LED frame has been sent. You can use this to give a **Semaphore** back to your LED task so it knows the hardware is ready for the next frame, which is much more efficient than polling `IsTransmitterBusy`.
